Here is an optimized relational schema modeled for PostgreSQL and Supabase. The setup uses strict foreign key constraints, JSONB metadata payloads for unstructured context, and vector embeddings on episodic memories to empower continuous semantic retrieval.

The memory system decouples discrete transactional chat logs from long-term factual statements, utilizing pgvector for cosine similarity queries across your workspace.

schema.sql Postgres 16 + pgvector 
-- Enable vector extension for associative memories
create extension if not exists vector with schema extensions;

-- 1. Conversation threads container
create table public.conversations (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  title text not null default 'New Conversation',
  pinned boolean default false,
  metadata jsonb default '{}'::jsonb,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- 2. Granular dialogue turns
create table public.messages (
  id uuid primary key default gen_random_uuid(),
  conversation_id uuid not null references public.conversations(id) on delete cascade,
  role text not null check (role in ('user', 'assistant', 'system')),
  content text not null,
  token_count integer,
  citations jsonb default '[]'::jsonb,
  created_at timestamptz not null default now()
);

-- 3. Episodic recall memory store
create table public.memories (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  fact text not null,
  category text check (category in ('preference', 'project', 'instruction')),
  embedding vector(1536),
  confidence real default 0.95,
  last_accessed_at timestamptz default now(),
  created_at timestamptz not null default now()
);

-- Index embeddings for sub-millisecond retrieval
create index on public.memories using hnsw (embedding vector_cosine_ops);
